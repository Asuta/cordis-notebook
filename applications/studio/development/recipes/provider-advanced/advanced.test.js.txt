import { test } from 'node:test'
import assert from 'node:assert/strict'
import { createRemoteProvider, accountId } from '../advanced/provider.js'
import { createTransport } from '../advanced/transport.js'
import { startMock } from '../advanced/mock-server.js'
const input = () => ({ model: 'sample-image', prompt: '原始要求', assetIds: [], parameters: { steps: 3 } })
test('真实本机 HTTP：目录、参数、输入快照、先落盘 ID 再轮询、恢复不重复提交', async t => {
  const mock = await startMock({ polls: 1 }); t.after(mock.close)
  let config = { baseUrl: mock.origin, apiKey: mock.key }, remote
  const provider = createRemoteProvider({ id: 'test', label: '测试', config: { private: () => ({ ...config }), public: () => ({ hasKey: true }) }, transport: createTransport({ pollInterval: 1 }) })
  assert.equal((await provider.listModels({ query: 'sample', limit: 1 })).items[0].id, 'sample-image')
  assert.equal((await provider.listModels({ kind: 'video' })).total, 0)
  const model = await provider.getModel('sample-image'); model.defaults.steps = 99
  assert.equal((await provider.getModel('sample-image')).defaults.steps, 2)
  for (const patch of [{ parameters: { steps: 0 } }, { parameters: { unknown: 1 } }, { assetIds: ['a'] }, { model: 'unknown' }]) assert.throws(() => provider.prepare({ ...input(), ...patch }))
  const source = input(), prepared = provider.prepare(source); source.prompt = '后续修改'; source.parameters.steps = 1
  const original = config; config = { ...config, apiKey: 'later-key' }
  const checkpoints = [], signal = new AbortController().signal
  const output = await prepared.generate([], signal, { checkpoint: async value => {
    checkpoints.push(value); remote = value
    if (value.phase === 'submitted') assert.equal(mock.requests.filter(row => row.method === 'GET').length, 0)
  } })
  assert.deepEqual(checkpoints.map(value => value.phase), ['preparing', 'submitting', 'submitted'])
  assert.equal(checkpoints[0].account, accountId(original)); assert.equal(output[0].kind, 'image')
  assert.deepEqual(mock.requests[0].body, { model: 'sample-image', prompt: '原始要求', steps: 3 })
  const job = { provider: 'test', model: 'sample-image', outputKind: 'image', remote }
  assert.throws(() => provider.resume(job), /原服务账户/)
  config = original; await provider.resume(job).generate([], signal)
  assert.equal(mock.requests.filter(row => row.method === 'POST').length, 1)
  assert.ok(!JSON.stringify([provider.public(), prepared, remote]).includes(mock.key))
})
test('暂停后的迟到 ID 仍保存、取消停止轮询；缺 ID 不自动重试；远端失败终止', async t => {
  const mock = await startMock({ polls: 1 }); t.after(mock.close)
  const controller = new AbortController(), checkpoints = []
  const transport = createTransport({ pollInterval: 1, request: async (...args) => { const response = await fetch(...args); controller.abort(); return response } })
  const config = { baseUrl: mock.origin, apiKey: mock.key }
  await assert.rejects(transport.generate(config, input(), controller.signal, async value => checkpoints.push(value), accountId(config)), { name: 'AbortError' })
  assert.equal(checkpoints.at(-1).phase, 'submitted'); assert.equal(mock.requests.filter(row => row.method === 'GET').length, 0)
  mock.state.mode = 'uncertain'
  const normal = createTransport({ pollInterval: 1 }), remote = []
  await assert.rejects(normal.generate(config, input(), new AbortController().signal, async value => remote.push(value), accountId(config)), /缺少任务 ID/)
  assert.equal(remote.at(-1).phase, 'submitting'); assert.equal(mock.requests.filter(row => row.method === 'POST').length, 2)
  mock.state.mode = 'failed'
  await assert.rejects(normal.poll(config, 'task-1', new AbortController().signal), { terminal: true })
  const stopped = new AbortController(); stopped.abort()
  await assert.rejects(normal.poll(config, 'task-1', stopped.signal), { name: 'AbortError' })
})

test('响应格式、大小和读取失败不暴露上游内容或密钥', async () => {
  const config = { baseUrl: 'https://example.invalid', apiKey: 'fixture-secret' }, signal = new AbortController().signal
  for (const [request, message] of [
    [async () => new Response('null'), /响应必须为对象/],
    [async () => new Response('fixture-secret'), /无效 JSON/],
    [async () => new Response('x'.repeat(2 * 1024 * 1024 + 1)), /超过 2 MB/],
    [async () => ({ ok: true, body: { async *[Symbol.asyncIterator]() { throw Error('fixture-secret') } } }), /响应读取失败/],
    [async () => { throw Error('fixture-secret') }, /连接失败/],
  ]) {
    const transport = createTransport({ request })
    await assert.rejects(transport.poll(config, 'task-1', signal), error => {
      assert.match(error.message, message); assert.ok(!error.message.includes(config.apiKey)); return true
    })
  }
})
